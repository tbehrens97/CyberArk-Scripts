STEP 3 - STAGE THE SWT NATIVE DLLs
-------------------------------------------------------------------------------

Add-Type -AssemblyName System.IO.Compression.FileSystem

$dest   = 'C:\SAP4HANA'
$staged = 0

Get-ChildItem "$dest\plugins" -Filter 'org.eclipse.swt.win32.win32.x86_64_*.jar' | ForEach-Object {
    $zip = [System.IO.Compression.ZipFile]::OpenRead($_.FullName)
    try {
        foreach ($entry in $zip.Entries) {
            if ($entry.Name -like '*.dll') {
                $target = Join-Path $dest $entry.Name
                [System.IO.Compression.ZipFileExtensions]::ExtractToFile($entry, $target, $true)
                Write-Host "Staged $($entry.Name)"
                $staged++
            }
        }
    } finally { $zip.Dispose() }
}

if ($staged -eq 0) { Write-Warning 'No SWT natives found - check C:\SAP4HANA\plugins.' }

# Verify (expect swt-win32-*, swt-gdip-*, swt-wgl-*, swt-osversion-win32-*)
Get-ChildItem 'C:\SAP4HANA' -Filter 'swt-*.dll' | Select-Object Name, Length


-------------------------------------------------------------------------------
STEP 4 - STAGE THE JNA NATIVE DLL (searches nested jars)
-------------------------------------------------------------------------------

Add-Type -AssemblyName System.IO.Compression.FileSystem

$dest   = 'C:\SAP4HANA'
$target = Join-Path $dest 'jnidispatch.dll'
$staged = $false

function Find-JniDispatch {
    param([System.IO.Compression.ZipArchive]$Zip, [string]$Origin)
    foreach ($e in $Zip.Entries) {
        if ($e.Name -eq 'jnidispatch.dll' -and $e.FullName -match 'win32-x86-64|win32-amd64|win32-x86_64') {
            [pscustomobject]@{ Entry = $e; Origin = $Origin; Path = $e.FullName }
        }
        elseif ($e.Name -like '*.jar') {
            $ms = New-Object System.IO.MemoryStream
            $s = $e.Open(); $s.CopyTo($ms); $s.Close(); $ms.Position = 0
            try {
                $inner = New-Object System.IO.Compression.ZipArchive($ms, 'Read')
                Find-JniDispatch -Zip $inner -Origin "$Origin!$($e.FullName)"
            } catch { }
        }
    }
}

Get-ChildItem $dest -Recurse -Filter '*.jar' -ErrorAction SilentlyContinue | ForEach-Object {
    if ($staged) { return }
    try {
        $zip = [System.IO.Compression.ZipFile]::OpenRead($_.FullName)
        try {
            $hit = Find-JniDispatch -Zip $zip -Origin $_.FullName | Select-Object -First 1
            if ($hit) {
                [System.IO.Compression.ZipFileExtensions]::ExtractToFile($hit.Entry, $target, $true)
                Write-Host "Staged jnidispatch.dll from $($hit.Origin) ($($hit.Path))"
                $staged = $true
            }
        } finally { $zip.Dispose() }
    } catch { }
}

if (-not $staged) { Write-Warning 'jnidispatch.dll not found - use STEP 4b.' }

-------------------------------------------------------------------------------
STEP 7 - PATCH hdbstudio.ini
Backs up first. Inserts -pluginCustomization before -vmargs and appends the
-D properties at the end. Safe to re-run (skips entries already present).
Does NOT add -Duser.home or -configuration; the dispatcher sets those per session.
-------------------------------------------------------------------------------

$ini = 'C:\SAP4HANA\hdbstudio.ini'
Copy-Item $ini "$ini.bak-$(Get-Date -Format yyyyMMdd-HHmmss)" -Force

$lines = [System.Collections.Generic.List[string]](Get-Content $ini)

if ($lines -notcontains '-pluginCustomization') {
    $idx = $lines.IndexOf('-vmargs')
    if ($idx -lt 0) { throw 'No -vmargs line found in hdbstudio.ini' }
    $lines.Insert($idx, 'C:/SAP4HANA/plugin_customization.ini')
    $lines.Insert($idx, '-pluginCustomization')
}

foreach ($prop in @('-Dswt.library.path=C:\SAP4HANA','-Djna.boot.library.path=C:\SAP4HANA')) {
    if ($lines -notcontains $prop) { $lines.Add($prop) }
}

$lines | Set-Content $ini -Encoding ASCII
Get-Content $ini
